  3. Cd to the profile folder and create user-overrides.js in the profile folder:

// =======================
// Clearnet Hardened Firefox Profile - Arkenfox Overrides
// Goal:
// - DuckDuckGo default search via policies.json, not user.js
// - No search suggestions / URL-bar suggestions / form history
// - Blank new tabs and blank home page
// - Restore previous open tabs + pinned tabs
// - Clear cookies/storage globally, keep trusted logins via site exceptions
// - Hardware access only by user consent
// - Stronger privacy without breaking normal clearnet browsing too much
// =======================


// ======================================================
// 1. Background network checks
// ======================================================

user_pref("network.captive-portal-service.enabled", false);
user_pref("network.connectivity-service.enabled", false);


// ======================================================
// 2. Address bar / search behavior
// ======================================================

// Keep URL-bar search enabled for DuckDuckGo via policies.json.
user_pref("keyword.enabled", true);

// Avoid DNS guessing / single-word fixups.
user_pref("browser.fixup.dns_first_for_single_words", false);
user_pref("browser.fixup.alternate.enabled", false);

// Disable search suggestions.
user_pref("browser.search.suggest.enabled", false);
user_pref("browser.search.suggest.enabled.private", false);
user_pref("browser.urlbar.suggest.searches", false);
user_pref("browser.urlbar.suggest.engines", false);
user_pref("browser.urlbar.showSearchSuggestionsFirst", false);

// Disable Firefox Suggest / Quick Suggest / sponsored / trending.
user_pref("browser.urlbar.quicksuggest.enabled", false);
user_pref("browser.urlbar.suggest.quicksuggest.all", false);
user_pref("browser.urlbar.suggest.quicksuggest.sponsored", false);
user_pref("browser.urlbar.suggest.trending", false);

// Disable URL-bar suggestions from local data.
user_pref("browser.urlbar.suggest.history", false);
user_pref("browser.urlbar.suggest.bookmark", false);
user_pref("browser.urlbar.suggest.openpage", false);
user_pref("browser.urlbar.suggest.topsites", false);
user_pref("browser.urlbar.autoFill", false);

// Avoid speculative URL-bar connections.
user_pref("browser.urlbar.speculativeConnect.enabled", false);

// Show full URLs.
user_pref("browser.urlbar.trimURLs", false);


// ======================================================
// 3. Form/search history
// ======================================================

user_pref("browser.formfill.enable", false);


// ======================================================
// 4. Startup / new tab / home page
// ======================================================

// 3 = restore previous session.
// Required for previous tabs and pinned tabs.
user_pref("browser.startup.page", 3);

// Blank home page / new window fallback.
user_pref("browser.startup.homepage", "about:blank");

// Blank new tab.
user_pref("browser.newtabpage.enabled", false);
user_pref("browser.newtabpage.activity-stream.enabled", false);
user_pref("browser.newtab.preload", false);


// ======================================================
// 5. Session restore behavior
// ======================================================

// Keep crash/session restore enabled.
user_pref("browser.sessionstore.resume_from_crash", true);

// Store only current page per tab, not long back/forward chains.
user_pref("browser.sessionstore.max_serialize_back", 0);
user_pref("browser.sessionstore.max_serialize_forward", 0);

// Do not keep closed-tab/window undo history.
user_pref("browser.sessionstore.max_tabs_undo", 0);
user_pref("browser.sessionstore.max_windows_undo", 0);

// Session data write interval.
user_pref("browser.sessionstore.interval", 15000);


// ======================================================
// 6. Passwords / logins
// ======================================================

// Keep Firefox password manager.
user_pref("signon.rememberSignons", true);

// Do not autofill saved credentials automatically.
user_pref("signon.autofillForms", false);


// ======================================================
// 7. Shutdown cleanup
// ======================================================

// Keep sanitize-on-shutdown enabled.
user_pref("privacy.sanitize.sanitizeOnShutdown", true);

// Critical: do NOT clear history/session data,
// otherwise previous-session restore can break.
user_pref("privacy.clearOnShutdown.history", false);
user_pref("privacy.clearOnShutdown_v2.historyFormDataAndDownloads", false);
user_pref("privacy.clearHistory.historyFormDataAndDownloads", false);

// Do NOT clear site settings; cookie exceptions live there.
user_pref("privacy.clearOnShutdown.siteSettings", false);
user_pref("privacy.clearOnShutdown_v2.siteSettings", false);

// Clear cookies/storage globally.
// Trusted logins should survive via cookie exceptions.
user_pref("privacy.clearOnShutdown.cookies", true);
user_pref("privacy.clearOnShutdown_v2.cookiesAndStorage", true);

// Clear cache and offline app data.
user_pref("privacy.clearOnShutdown.cache", true);
user_pref("privacy.clearOnShutdown_v2.cache", true);
user_pref("privacy.clearOnShutdown.offlineApps", true);

// Clear downloads/form data on shutdown.
user_pref("privacy.clearOnShutdown.downloads", true);
user_pref("privacy.clearOnShutdown.formdata", true);


// ======================================================
// 8. Tracking protection / HTTPS / attribution
// ======================================================

// Strong Enhanced Tracking Protection.
// Usually good UX/privacy balance.
user_pref("browser.contentblocking.category", "strict");

// HTTPS-Only Mode.
// May need manual exceptions for old local devices / legacy HTTP sites.
user_pref("dom.security.https_only_mode", true);
user_pref("dom.security.https_only_mode_pbm", true);

// Reduce cross-origin referrer leakage.
user_pref("network.http.referer.XOriginTrimmingPolicy", 2);

// Disable privacy-invasive attribution/reporting surfaces.
user_pref("dom.private-attribution.submission.enabled", false);


// ======================================================
// 9. Geolocation / telemetry / experiments
// ======================================================

// Block geolocation by default.
user_pref("permissions.default.geo", 2);
user_pref("geo.enabled", false);
user_pref("geo.provider.network.url", "");

// Disable Firefox telemetry / studies / experiments.
user_pref("datareporting.healthreport.uploadEnabled", false);
user_pref("datareporting.policy.dataSubmissionEnabled", false);
user_pref("toolkit.telemetry.enabled", false);
user_pref("toolkit.telemetry.unified", false);
user_pref("toolkit.telemetry.archive.enabled", false);
user_pref("app.shield.optoutstudies.enabled", false);
user_pref("app.normandy.enabled", false);
user_pref("app.normandy.api_url", "");


// ======================================================
// 10. Google Safe Browsing minimization
// ======================================================

// Privacy gain: fewer Google-related Safe Browsing calls.
// Security cost: weaker malware/phishing/download protection.
// Keep this block if your priority is "no Google-related network services".
user_pref("browser.safebrowsing.malware.enabled", false);
user_pref("browser.safebrowsing.phishing.enabled", false);
user_pref("browser.safebrowsing.downloads.enabled", false);
user_pref("browser.safebrowsing.downloads.remote.enabled", false);
user_pref("browser.safebrowsing.provider.google4.dataSharing.enabled", false);
user_pref("browser.safebrowsing.provider.google4.updateURL", "");
user_pref("browser.safebrowsing.provider.google4.gethashURL", "");
user_pref("browser.safebrowsing.provider.google.updateURL", "");
user_pref("browser.safebrowsing.provider.google.gethashURL", "");


// ======================================================
// 11. Hardware / device access
// ======================================================

// 0 = ask, 1 = allow, 2 = block.
user_pref("permissions.default.camera", 0);
user_pref("permissions.default.microphone", 0);
user_pref("permissions.default.screen", 0);

// Block noisy/risky permission types.
user_pref("permissions.default.desktop-notification", 2);
user_pref("permissions.default.xr", 2);

// Autoplay off.
user_pref("media.autoplay.default", 5);
user_pref("media.autoplay.blocking_policy", 2);

// Minimize device/sensor APIs.
user_pref("dom.webmidi.enabled", false);
user_pref("dom.gamepad.enabled", false);
user_pref("device.sensors.enabled", false);
user_pref("dom.vibrator.enabled", false);

// WebRTC: keep video calls usable, but limit local IP exposure.
user_pref("media.peerconnection.ice.default_address_only", true);
user_pref("media.peerconnection.ice.no_host", true);


// ======================================================
// 12. Optional integrations / services
// ======================================================

// Disable Pocket.
user_pref("extensions.pocket.enabled", false);

// Disable Firefox Account / Sync.
// If you use Firefox Sync, remove this line.
user_pref("identity.fxaccounts.enabled", false);

// Disable DRM.
// If you use Netflix, Spotify Web Player, some paid streaming sites,
// remove or set this to true.
user_pref("media.eme.enabled", false);
